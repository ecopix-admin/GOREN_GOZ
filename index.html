// Açıq mənşəli Avto-Crawler Boti const fs = require('fs'); const https = require('https'); // İndekslənəcək ilkin saytlar və qaynaqlar (Seed URLs) const targetSites = [ { url: 'https://az.wikipedia.org/wiki/Avtomobil', category: 'Məlumat' } ]; let localDatabase = []; // Saytın mətnini çəkən funksiya function scrapeSite(site) { console.log(`[BOT] Sayt taranır: ${site.url}`); https.get(site.url, (res) => { let html = ''; res.on('data', chunk => html += chunk); res.on('end', () => { // HTML teqlərini təmizləyirik və mətni çıxarırıq const textContent = html.replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' '); // Avto ehtiyat hissələri və usta açar sözlərini analiz edirik const keywords = ['mühərrik', 'yağ', 'təkər', 'akümulyator', 'ustası', 'ehtiyat hissəsi', 'bamper', 'korobka']; const foundKeywords = keywords.filter(word => textContent.toLowerCase().includes(word)); const entry = { title: `Axtarış Tapıntısı: ${site.url}`, url: site.url, snippet: textContent.substring(0, 200) + '...', keywords: foundKeywords, timestamp: new Date().toISOString() }; localDatabase.push(entry); // Məlumatı yerli JSON faylına saxlayırıq (P2P Paylaşım üçün) fs.writeFileSync('auto_index.json', JSON.stringify(localDatabase, null, 2)); console.log('[BOT] Məlumat uğurla bazaya yazıldı.'); }); }).on('error', (err) => { console.error(`[XƏTA] ${site.url} taranarkən xəta baş verdi:`, err.message); }); } // Boti işə salırıq targetSites.forEach(scrapeSite);
